## Raney引理
若整数序列每一项不超过 1 ，总和为正整数 n ，则它的 m 个循环移位中恰有 n 个移位的 所有非空前缀和为正。
```cpp

#include <vector>
using namespace std;
using ll = long long;

const ll MOD = 998244353;
const int MAXN = 2e6;

vector<ll> fac(MAXN + 1), invfac(MAXN + 1);

// 快速幂
ll qpow(ll a, ll b) {
    ll res = 1;
    a %= MOD;
    while (b) {
        if (b & 1) res = res * a % MOD;
        a = a * a % MOD;
        b >>= 1;
    }
    return res;
}

// 预处理阶乘与阶乘逆元
void preprocess() {
    fac[0] = 1;
    for (int i = 1; i <= MAXN; ++i)
        fac[i] = fac[i - 1] * i % MOD;

    invfac[MAXN] = qpow(fac[MAXN], MOD - 2);          // 费马小定理求逆元
    for (int i = MAXN - 1; i >= 0; --i)
        invfac[i] = invfac[i + 1] * (i + 1) % MOD;
}

// 组合数 C(n,k) = n! / (k! (n-k)!)
ll C(int n, int k) {
    if (k < 0 || k > n) return 0;
    return fac[n] * invfac[k] % MOD * invfac[n - k] % MOD;   // 边乘边取模，防止溢出
}

int main() {
    preprocess();
    int T;
    scanf("%d", &T);
    while (T--) {
        int n, m, c, a, b;
        scanf("%d %d %d %d %d", &n, &m, &c, &a, &b);

        // 无解情况：钱不够买 m 瓶，或获奖次数不是整数
        if (m < n || (m - n) % c != 0) {
            printf("0\n");
            continue;
        }

        int k = (m - n) / c;                     // 必须获奖的次数
        ll inv_b = qpow(b, MOD - 2);             // b 的逆元
        ll p = a * inv_b % MOD;                  // 获奖概率 = a/b
        ll q = (b - a) * inv_b % MOD;            // 未获奖概率 = (b-a)/b
        ll inv_m = qpow(m, MOD - 2);             // m 的逆元

        // 答案 = (n/m) * C(m, k) * p^k * q^{m-k}
        ll base = n * inv_m % MOD * C(m, k) % MOD;
        ll ans = base * qpow(p, k) % MOD * qpow(q, m - k) % MOD;
        printf("%lld\n", ans);
    }
    return 0;
}
```

## 裴属定理
**定义**
- 设 $a,b$ 是不全为零的整数．那么，对于任意整数 $x,y$，都有 $\gcd(a,b) \mid ax+by$ 成立；而且，存在整数 $x,y$，使得 $ax+by=\gcd(a,b)$ 成立．

## 模逆元

**定义**
- 在模运算中，对于整数 $a$ 和模数 $m$，若存在整数 $x$ 使得：
$$
a \times x \equiv 1 \pmod{m}
$$
则称 $x$ 是 $a$ 模 $m$ 的**逆元**，记作 $a^{-1} \bmod m$。

**定理**
- 当且仅当a与m互质时$a$ 模 $m$ 的逆元存在（由贝祖等式：存在 $x,y$ 使 $ax + my = 1$，两边模 $m$ 得 $ax \equiv 1 \pmod{m}$）
**求法**
- 扩展欧几里得
```cpp
ll mie1(ll a, ll m) {  
    ll x, y;  
    exgcd(a, m, x, y);  
    return (x % m + m) % m;  
}
```
- 费马小定理 **（若 m 是质数，且整数 a 不是 m 的倍数才能用）**
```cpp
ll mie2(ll a, ll m) {  
    return ppow(a, m - 2, m);  
}
```
 
## 同余
- **定义**
  - 若整数 $a$ 与 $b$ 满足 $m \mid (a-b)$（即 $m$ 整除 $a-b$），则称 $a$ 与 $b$ 模 $m$ 同余，记作  
$a \equiv b \pmod{m}$。

- **性质**
	- 自反性：$a \equiv a \pmod{m}$
	- 对称性：若 $a \equiv b \pmod{m}$，则 $b \equiv a \pmod{m}$
	- 传递性：若 $a \equiv b \pmod{m},\ b \equiv c \pmod{m}$，则 $a \equiv c \pmod{m}$
	
	- 线性运算（加减）：若 $a,b,c,d \in \mathbf{Z},\ m \in \mathbf{N}^*,\ a \equiv b \pmod{m},\ c \equiv d \pmod{m}$，则  
	  $a \pm c \equiv b \pm d \pmod{m}$
	- 线性运算（乘）：若 $a,b,c,d \in \mathbf{Z},\ m \in \mathbf{N}^*,\ a \equiv b \pmod{m},\ c \equiv d \pmod{m}$，则  
	  $a \times c \equiv b \times d \pmod{m}$
	- 同乘非零整数：若 $a,b \in \mathbf{Z},\ k,m \in \mathbf{N}^*,\ a \equiv b \pmod{m}$，则 $ak \equiv bk \pmod{mk}$
	- 同除（约去公因子）：若 $a,b \in \mathbf{Z},\ d,m \in \mathbf{N}^*,\ d \mid a,\ d \mid b,\ d \mid m$，且 $a \equiv b \pmod{m}$，则  
	  $\dfrac{a}{d} \equiv \dfrac{b}{d} \pmod{\dfrac{m}{d}}$
	- 模数缩小：若 $a,b \in \mathbf{Z},\ d,m \in \mathbf{N}^*,\ d \mid m$，且 $a \equiv b \pmod{m}$，则 $a \equiv b \pmod{d}$
	- 最大公约数相等：若 $a,b \in \mathbf{Z},\ m \in \mathbf{N}^*$，且 $a \equiv b \pmod{m}$，则 $(a,m) = (b,m)$。  
	  若 $d$ 能整除 $m$ 及 $a,b$ 中的一个，则 $d$ 必定能整除 $a,b$ 中的另一个。
- **消去律**
	- 若 $ac \equiv bc \pmod{m}$，则 $a \equiv b \pmod{m / \gcd(m,c)}$。
### 同余类和剩余系
- **同余类（剩余类）**  
	- 固定模数 m，把所有整数按除以 m 的余数分成 m 类，每一类叫一个同余类（剩余类）
	  例：模 3 有 3 个同余类：{...,0,3,6,...},{...,1,4,7,...},{...,2,5,8,...}
- **完全剩余系**  
- 从每个同余类里各取一个代表，得到 m 个数，叫模 m 的完全剩余系
  常见取法：{0,1,2,...,m−1}。
- **简化剩余系（既约剩余系）**  
	- 只取那些与 m 互质的同余类（共 φ(m) 个），每类取一个代表，组成简化剩余系。  
	  例：模 8 的简化剩余系可以是 {1,3,5,7}
### 线性同余方程
$$ax = b(\bmod m)$$
1. 计算 $d = \gcd(a, m)$。
2. 若 $d \nmid b$，无解，否则解的数量为$d$。
3. 令 $a' = a/d,\ b' = b/d,\ m' = m/d$。
4. 用扩展欧几里得求 $a'$ 模 $m'$ 的逆元 $inv$。
5. 计算 $x_0 = (inv \cdot b') \bmod m'$。
6. 全部解为 $x = x_0 + k \cdot m' \pmod{m},\ k = 0,1,\dots,d-1$。

## 中国剩余定理
**问题形式**
求解如下同余方程组（$n$ 个方程）：
$$
\begin{cases}
x \equiv a_1 \pmod{m_1} \\
x \equiv a_2 \pmod{m_2} \\
\quad\ \ \vdots \\
x \equiv a_n \pmod{m_n}
\end{cases}
$$
其中**模数 $m_1, m_2, \dots, m_n$ 不一定两两互质**，要求输出最小非负解 $x$，或者判断无解。
**CRT**
  - 条件：模数 $m_1,\dots,m_k$ 两两互质  
  - 令 $M = \prod m_i$  
  - 对每个 $i$：  
    - $M_i = M / m_i$  
    - 求 $t_i \equiv M_i^{-1} \pmod{m_i}$  
  - 解：$x = \sum_{i=1}^k a_i M_i t_i \bmod M$  
```cpp
typedef long long ll;
ll exgcd(ll a, ll b, ll &x, ll &y) {
	if (!b) { x = 1; y = 0; return a; }
	ll d = exgcd(b, a % b, y, x);
	y -= a / b * x;
	return d;
}
ll crt(const vector<ll>& a, const vector<ll>& m) {
	ll M = 1, ans = 0;
	for (ll mi : m) M *= mi;
	for (size_t i = 0; i < a.size(); ++i) {
		ll Mi = M / m[i];
		ll ti, y;
		exgcd(Mi, m[i], ti, y);
		ti = (ti % m[i] + m[i]) % m[i];
		ans = (ans + (__int128)a[i] * Mi % M * ti) % M;
	}
	return (ans % M + M) % M;
}
```

## 异或线性基
```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

struct LinearBasis {
    static const int MAXL = 62;        // 处理非负整数时使用 62 位（0..62），可根据需要调整
    ll base[MAXL + 1];                // 基向量：base[i] 的最高位为 i，且该列已对角化
    bool zero;                        // 标记 0 是否可由已有向量线性表示

    LinearBasis() {
        memset(base, 0, sizeof(base));
        zero = false;
    }

    // 插入一个向量 x，贪心 + 对角化
    void insert(ll x) {
        for (int i = MAXL; i >= 0; --i) {
            if (!(x >> i & 1)) continue;          // 若当前位为 0，跳过
            if (base[i]) {                        // 该位已有基向量，用其消去当前位的 1
                x ^= base[i];
            } else {
                for (int j = i - 1; j >= 0; --j)
                    if (x >> j & 1) x ^= base[j];
                for (int j = i + 1; j <= MAXL; ++j)
                    if (base[j] >> i & 1) base[j] ^= x;
                base[i] = x;
                return;
            }
        }
        // 若 x 被消为 0，说明它可由现有基线性表示
        zero = true;
    }
    ll queryMax() {
        ll res = 0;
        for (int i = MAXL; i >= 0; --i)
            if ((res ^ base[i]) > res) res ^= base[i];
        return res;
    }
    // 查询非零最小异或和
    ll queryMin() {
        if (zero) return 0;                // 若 0 可表示，最小即为 0
        for (int i = 0; i <= MAXL; ++i)
            if (base[i]) return base[i];   // 因已对角化，最低位的基就是最小非零值
        return 0;                          // 空基
    }
    // 查询第 k 小异或和（k 从 1 开始），若 k 超出范围返回 -1
    ll queryKth(unsigned long long k) {   // 使用 unsigned long long 避免溢出
        if (zero) {
            if (k == 1) return 0;         // 第 1 小就是 0
            --k;                          // 去除 0 后，在非零子集中查找
        }
        vector<ll> bases;                 // 收集所有非零基（已对角化，最高位互异）
        for (int i = 0; i <= MAXL; ++i)
            if (base[i]) bases.push_back(base[i]);

        int sz = bases.size();
        if (k > (1ULL << sz)) return -1;  // 超出可表示的不同异或和的数目

        ll res = 0;
        for (int i = 0; i < sz; ++i)
            if (k >> i & 1) res ^= bases[i];
        return res;
    }
};
```
## 容斥原理与欧拉函数
```cpp
#include <iostream>
#include <vector>
using namespace std;

// 公式法
int phi_formula(int n) {
    int res = n;
    for (int i = 2; i * i <= n; ++i) {
        if (n % i == 0) {
            res = res / i * (i - 1);
            while (n % i == 0) n /= i;
        }
    }
    if (n > 1) res = res / n * (n - 1);
    return res;
}

// 显式容斥法
vector<int> getPrimeFactors(int n) {
    vector<int> primes;
    for (int i = 2; i * i <= n; ++i) {
        if (n % i == 0) {
            primes.push_back(i);
            while (n % i == 0) n /= i;
        }
    }
    if (n > 1) primes.push_back(n);
    return primes;
}

int phi_ie(int n) {
    if (n == 1) return 1;
    vector<int> p = getPrimeFactors(n);
    int k = p.size(), res = n;
    for (int mask = 1; mask < (1 << k); ++mask) {
        int prod = 1, cnt = 0;
        for (int i = 0; i < k; ++i) {
            if (mask & (1 << i)) {
                prod *= p[i];
                ++cnt;
            }
        }
        if (cnt & 1) res -= n / prod;
        else         res += n / prod;
    }
    return res;
}

int main() {
    for (int n : {1, 2, 6, 12, 30, 100}) {
        cout << "φ(" << n << ") = "
             << phi_formula(n) << " (公式法), "
             << phi_ie(n) << " (容斥法)\n";
    }
    return 0;
}
```

```cpp
#include <cstdio>
#include <vector>

using ll = long long;

struct Node {
    int l, r;
    ll sum, tag;
};

std::vector<Node> tr;
std::vector<ll> a;

// 左右儿子编号
inline int ls(int p) { return p << 1; }
inline int rs(int p) { return p << 1 | 1; }

// 用左右儿子的信息更新当前节点
void pushup(int p) {
    tr[p].sum = tr[ls(p)].sum + tr[rs(p)].sum;
}

// 下传懒标记
void pushdown(int p) {
    if (tr[p].tag) {
        ll t = tr[p].tag;
        int lc = ls(p), rc = rs(p);
        tr[lc].tag += t;
        tr[rc].tag += t;
        tr[lc].sum += (tr[lc].r - tr[lc].l + 1LL) * t;
        tr[rc].sum += (tr[rc].r - tr[rc].l + 1LL) * t;
        tr[p].tag = 0;
    }
}

// 建树
void build(int p, int l, int r) {
    tr[p] = {l, r, a[l], 0};
    if (l == r) return;
    int mid = (l + r) >> 1;
    build(ls(p), l, mid);
    build(rs(p), mid + 1, r);
    pushup(p);
}

// 区间加法
void update(int p, int l, int r, ll val) {
    if (l <= tr[p].l && tr[p].r <= r) {
        tr[p].sum += (tr[p].r - tr[p].l + 1LL) * val;
        tr[p].tag += val;
        return;
    }
    pushdown(p);
    int mid = (tr[p].l + tr[p].r) >> 1;
    if (l <= mid) update(ls(p), l, r, val);
    if (r > mid)  update(rs(p), l, r, val);
    pushup(p);
}

// 区间求和
ll query(int p, int l, int r) {
    if (l <= tr[p].l && tr[p].r <= r) {
        return tr[p].sum;
    }
    pushdown(p);
    int mid = (tr[p].l + tr[p].r) >> 1;
    ll res = 0;
    if (l <= mid) res += query(ls(p), l, r);
    if (r > mid)  res += query(rs(p), l, r);
    return res;
}
```

## 伸展树
```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;

struct Node {
    int ch[2];   // 左右儿子下标，0 表示空
    int fa;      // 父节点下标
    int val;     // 节点存储的值
    int cnt;     // 该值出现的次数
    int sz;      // 子树中元素总个数（含重复计数）
} tr[MAXN];

int root, tot;

// 更新节点 x 的子树大小
void pushup(int x) {
    tr[x].sz = tr[tr[x].ch[0]].sz + tr[tr[x].ch[1]].sz + tr[x].cnt;
}

// 旋转 x，x 是其父节点的某个儿子
void rotate(ll x) {  
    ll y = tr[x].fa, z = tr[y].fa;  
    ll k = tr[y].s[1] == x;  
    tr[y].s[k] = tr[x].s[k ^ 1];  
    if (tr[x].s[k ^ 1])tr[tr[x].s[k ^ 1]].fa = y;  
    tr[x].s[k ^ 1] = y;  
    tr[y].fa = x;  
    if (z)tr[z].s[tr[z].s[1] == y] = x;  
    tr[x].fa = z;  
    pushup(y), pushup(x);  
}

// 将 x 伸展到 goal 的儿子位置，goal == 0 表示伸展到根
void splay(int x, int goal) {
    while (tr[x].fa != goal) {
        int y = tr[x].fa, z = tr[y].fa;
        if (z != goal) {
            // 判断是否为同一方向，决定是单旋还是双旋
            if ((tr[y].ch[1] == x) ^ (tr[z].ch[1] == y)) rotate(x);
            else rotate(y);
        }
        rotate(x);
    }
    if (goal == 0) root = x;
}

// 查找值为 v 的节点并伸展到根，若不存在则伸展最后访问的节点
void find(int v) {
    int u = root;
    while (tr[u].ch[v > tr[u].val] && tr[u].val != v)
        u = tr[u].ch[v > tr[u].val];
    splay(u, 0);
}

// 插入值 v，如果已存在则增加计数，否则新建节点并伸展到根
void insert(int v) {
    int u = root, fa = 0;
    while (u && tr[u].val != v) {
        fa = u;
        u = tr[u].ch[v > tr[u].val];
    }
    if (u) {
        tr[u].cnt++;
        pushup(u);
    } else {
        u = ++tot;
        if (fa) tr[fa].ch[v > tr[fa].val] = u;
        tr[u].fa = fa;
        tr[u].val = v;
        tr[u].cnt = tr[u].sz = 1;
    }
    splay(u, 0);
}

// 查找值 v 的严格前驱节点编号，并伸展到根
int pre(int v) {
    find(v);
    if (tr[root].val < v) return root;   // 根值已经小于 v，根即前驱
    int u = tr[root].ch[0];
    while (tr[u].ch[1]) u = tr[u].ch[1];
    splay(u, 0);
    return u;
}

// 查找值 v 的严格后继节点编号，并伸展到根
int nxt(int v) {
    find(v);
    if (tr[root].val > v) return root;   // 根值已经大于 v，根即后继
    int u = tr[root].ch[1];
    while (tr[u].ch[0]) u = tr[u].ch[0];
    splay(u, 0);
    return u;
}

// 删除值 v（若有多个只删除一个），需要树中已有两个哨兵节点 -INF 和 INF
void del(int v) {
    int p = pre(v);
    int q = nxt(v);
    splay(p, 0);
    splay(q, p);
    int u = tr[q].ch[0];           // 此时待删除节点一定是 q 的左儿子
    if (tr[u].cnt > 1) {
        tr[u].cnt--;
        splay(u, 0);               // 更新后伸展维护
    } else {
        tr[q].ch[0] = 0;           // 直接移除
        pushup(q); pushup(p);
    }
}

// 查询值 v 的排名（比 v 小的数的个数 + 1），假设 v 存在
int rk(int v) {
    find(v);
    return tr[tr[root].ch[0]].sz + 1;
}

// 查询第 k 小的值（k 从 1 开始），保证 k 合法，并伸展对应节点到根
int kth(int k) {
    int u = root;
    while (true) {
        int lsz = tr[tr[u].ch[0]].sz;
        if (k <= lsz)
            u = tr[u].ch[0];
        else if (k <= lsz + tr[u].cnt)
            break;
        else {
            k -= lsz + tr[u].cnt;
            u = tr[u].ch[1];
        }
    }
    splay(u, 0);
    return tr[u].val;
}

int main() {
    insert(INT_MIN);  // 左哨兵
    insert(INT_MAX);  // 右哨兵
    // 接下来可以自由调用 insert / del / rk / kth / pre / nxt 等

}
```

## 点分治
1. 找出树的重心做根，getroot ()
2. 求出子树中的各点到根的距离，getdis ()

## 左偏树-可并堆

性质1：堆的性质，对于任意节点 $v[x] \le v[lc[x]],v[rc[x]]$（小根堆）。
性质2：左偏性质，左儿子的距离 $\ge$ 右儿子的距离，$dis[lc] \ge dis[rc]$。
性质3：任意节点的距离 = 其右儿子的距离 + 1，$dis[x] = dis[rc] + 1$。
性质4：一棵有 $n$ 个节点的二叉树，根的 $dis \le \log(n + 1) - 1$。
因为根的距离为 $x$ 的二叉树至少有 $x + 1$ 层是满二叉树，那么至少有 $2^{x+1} - 1$ 个节点。
即 $n \ge 2^{x+1} - 1$，所以 $x \le \log(n + 1) - 1$。

```cpp
#include<iostream>  
#include <vector>  
using namespace std;  
using ll = long long;  
ll n = 1e5;  
  
//idealize持续保持备战ICPC Created in 2026/8/24 21:16  
struct node {  
    ll v, l, r, dis;  
};  
  
vector<bool> del(n);

ll merge(ll x, ll y) {  
    if (!x || !y) return x + y;  
    if (tr[x].v == tr[y].v ? x > y : tr[x].v > tr[y].v) swap(x, y);  
    tr[x].r = merge(tr[x].r, y);  
  
    if (tr[tr[x].l].dis < tr[tr[x].r].dis) swap(tr[x].l, tr[x].r);  
    tr[x].dis = tr[tr[x].r].dis + 1;  
    return x;  
}  
  
ll find(ll x) {  
    if (dsu[x] == x)return x;  
    return dsu[x] = find(dsu[x]);  
}  
  
void comb(ll x, ll y) {  
    if (find(x) == find(y) || del[x] || del[y])return;  
    dsu[dsu[x]] = dsu[dsu[y]] = merge(dsu[x], dsu[y]);  
}  
  
int main() {  
    ios::sync_with_stdio(false);  
    cin.tie(nullptr);
```

# 图论
## 最短路
### Dijkstra算法+堆优化
$O(n+m)$
```cpp
const int N = 1e5;  
vector<pair<ll, ll> > e[N + 1];  
vector<ll> d(N + 1, 1e15);  
bool vis[N + 1];  
ll n, m, s;  
  
void dijkstra(ll u) {  
    priority_queue<pair<ll, ll> > pq;  
    pq.push({0, u});  
    d[u] = 0;  
    while (!pq.empty()) {  
        u = pq.top().second;  
        pq.pop();  
        if (vis[u]) continue;  
        vis[u] = true;  
        for (auto [v,w]: e[u]) {  
            if (w + d[u] < d[v]) {  
                d[v] = w + d[u];  
                pq.push({-d[v], v});  
            }  
        }  
    }  
}
```
### Bellman-Ford算法+SPFA优化
$O(m \log m)$
```cpp
bool spfa(ll s, vector<vector<pair<ll, ll> > >& e, vector<ll> &d) {  
    vector<ll> cnt(n + 1);  
    vector<bool> vis(n + 1);  
    queue<ll> q;  
    q.push(s);  
    d[s] = 0;  
    while (!q.empty()) {  
        ll u = q.front();  
        q.pop();  
        vis[u] = false;  
        for (auto [v,w]: e[u]) {  
            if (d[v] > d[u] + w) {  
                d[v] = d[u] + w;  
                if (!vis[v]) {  
                    q.push(v);  
                    vis[v] = true;  
                }  
                cnt[v] = cnt[u] + 1;  
                if (cnt[v] >= n) return true;  
            }  
        }  
    }  
    return false;  
}
```

### Floyd算法

```cpp
void floyd() {  
    for (ll k = 1; k <= n; k++)  
        for (ll i = 1; i <= n; i++)  
            for (ll j = 1; j <= n; j++)  
                if (d[i][k] != 1e15 && d[k][j] != 1e15)   
                    d[i][j] = min(d[i][j], d[i][k] + d[k][j]);  
}
```

### Prim算法
$O(n^2)$
```cpp
bool prim(ll s) {  
    vector d(n + 1, INF);  
    vector<bool> vis(n + 1);  
    d[s] = 0;  
    ll cnt = 0;  
    for (int i = 1; i <= n; i++) {  
        ll u = 0;  
        for (int j = 1; j <= n; j++) if (!vis[j] && d[j] < d[u])u = j;  
        vis[u] = true;  
        ans += d[u];  
        if (d[u] != INF) cnt++;  
        for (auto [v,w]: e[u]) {  
            if (d[v] > w) d[v] = w;  
        }  
    }  
    return cnt == n;  
}
```
